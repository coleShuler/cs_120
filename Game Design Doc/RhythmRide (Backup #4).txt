        #Main Player icon
class Space_Key(simpleGE.Sprite):
    def __init__(self, scene):
        super().__init__(scene)
        self.current_image = "Light_Space.png"
        self.setImage(self.current_image)
        self.setSize(400, 75)
        self.position = (320, 400)
        self.key_active_until = 0
        self.key_pressed = False
        self.music_stopped = False
    def process(self):
        if not mixer.music.get_busy():
            self.music_stopped = True
        if self.music_stopped:
            return
        if self.isKeyPressed(pygame.K_SPACE) and not self.key_pressed:
            self.current_image = "Dark_Space.png"
            self.setImage(self.current_image)
            self.setSize(400, 75)
            self.position = (320, 400)
            self.key_active_until = pygame.time.get_ticks() + 100
            self.key_pressed = True
        if pygame.time.get_ticks() > self.key_active_until:
            self.current_image = "Light_Space.png"
            self.setImage(self.current_image)
            self.setSize(400, 75)
            self.position = (320, 400)
        if not self.isKeyPressed(pygame.K_SPACE):
            self.key_pressed = False

class PH(simpleGE.Sprite):
    def __init__(self, scene):
        super().__init__(scene)
        self.setImage("PlaceHolder.jpg")
        self.setSize(150, 150)
        self.position = (1, 1)
        self.win = False
    def process(self):
        if not pygame.mixer.music.get_busy():
            self.scene.beat1.dy = 0
            self.scene.beat2.dy = 0
            self.scene.beat3.dy = 0
            self.scene.beat5.dy = 0
            self.scene.beat1.setImage ("Hit.png")
            self.scene.beat1.setImage ("Hit.png")
            self.scene.beat2.setImage ("Hit.png")
            self.scene.beat3.setImage ("Hit.png")
            self.scene.beat4.setImage ("Hit.png")
            self.scene.beat5.setImage ("Hit.png")
            self.setImage("win.jpg")
            self.setSize(320, 240)
            self.position = (320, 240)
        if not self.win and self.scene.beat1.y > 470 and self.scene.beat1.current_image == "Beat.png":
            self.setImage("lose.jpg")
            self.setSize(320, 240)
            self.position = (320, 240)
            self.scene.beat4.dy = 0

class Beat1(simpleGE.Sprite):
    def __init__(self, scene):
        super().__init__(scene)
        self.current_image = "Beat.png"
        self.setImage(self.current_image)
        self.setSize(40, 40)
        self.position = (320, 75)
        self.dy = 0
 
#Title Screen Buffer
    def process(self):
        if self.scene.title.y > 0:
            self.dy = 0
        if self.scene.title.y > 700:
            self.dy = 6.05
 
#Check if hit            
        if self.collidesWith(self.scene.space_Key) and self.scene.space_Key.current_image == "Dark_Space.png":
            self.current_image = "Hit.png"
            self.setImage(self.current_image)
            self.setSize(20, 20)

#Check for ball to respawn
        if self.scene.beat1.y > 470 and self.scene.beat1.current_image == "Hit.png":
            self.current_image = "Beat.png"
            self.setImage(self.current_image)
            self.position = (320, 1)
            self.setSize(40, 40)

#Lost Game        
        if self.scene.beat1.y > 470 and self.scene.beat1.current_image == "Beat.png":
            self.dy = 0

class Game(simpleGE.Scene):
    def __init__(self):
        super().__init__()
        self.setImage("BG.jpg")
        self.title = Title(self)
        self.space_Key = Space_Key(self)
        self.beat1 = Beat1(self)
        self.beat2 = Beat2(self)
        self.beat3 = Beat3(self)
        self.beat4 = Beat4(self)
        self.beat5 = Beat5(self)
        self.ph = PH(self)
        self.sprites = [self.space_Key,
                        self.beat1,
                        self.beat2,
                        self.beat3,
                        self.beat4,
                        self.beat5,
                        self.ph,
                        self.title,]



#Runs Game Program
def main():
    game = Game()
    game.start()
if __name__ == "__main__":
    main()
